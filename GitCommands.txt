
	// --global се използва като настройки на всички проекти, по които работиш
	// даваш си username за гит профила
	git config --global user.name "username"
	
	// даваш си email за гит профила 
	git config --global user.email "username email"
	
	// слага ти notepad++ като основен едитор като трябва да пишеш комит съобщения
	git config --global core.editor "'C:/Program Files/Notepad++/notepad++.exe' -multiInst -notabbar -nosession -noPlugin"
	
	// показва ти всички конфигурирани настройки на git профила ти
	git config --list 
	
	// промяна на името на основния бранч, който имаш 
	git config --global init.defaultBranch main 
	
================================================

	// инициализира на git repo 
	git init repo 
	
	// клониране на git repo, когато се клонира url адреса ще има псевдоним origin
	git clone <url>
	
	// клониране на git repo, когато се клонира url адреса ще има псевдоним <newName>
	git clone <url> <newName>
	
================================================

	// показва ти информация относно бранча ти, stadging area-та ти и друга информация 
	git status
	
	// добавяне на файл/ове в stadging area-та 
	git add <fileName>
	git add . // това е за всички файлове
	
	// Съществува файл .gitignore, в който може да добавиш какви файлове да се игнорират
	
	// комитване на промените, които са в staging area-та
	git commit // ще ти отговори редактора дефиниран в git config code.editor за да напишеш commit съобщение
	git commit -m <commitMessage> //  директно ти комитва с <commitMessage>
	git commit -a -m <commitMessage> // комитва всички файлове, който са tracked и са били modified (не е нужно да са били в staging area) и ги комитва с <commitMessage>
	
	// ако си пропуснал да добавиш файл към последния комит и искаш да смениш съобщението
	git commit --amend
	git commit --amend --no-edit // ако искаш да не променяш комит съобщението 
	git commit --amend -m <newCommitMessage> // ако не ти се занимава с код едитор 
	
=================================================

	// изтрива <fileName> и добавя тази промяна в staging area-та 
	git rm <filename> 
	
	// ако <filename> има някакви tracked промени
	git rm -f <filename>
	
	// ако искаш да махнеш файла от stage-ing area-та, но да си остане на диска
	git rm --chached <fileName>
	
	// ако искаш да преименуваш файл 
	git mv <oldName> <newName>
	
====================================================

	// показва ти историята на git комитите (най-новите са най-отгоре), показва и също чексума, дата на комит, автора и друга информация
	git log 
	
	// освен, че ти показва същата информация като git log, тук ще ти опише и какви са били самите разлики
	git log -p или git log --patch
	
	// ако искаш да видиш просто броя на добавени и премахнати редове използвай
	git log --stat 
	
	// ако искаш да ти е по-красиво описано или да ти показва git историята като граф 
	git log --pretty // показва ти го по-красиво 
	git log --graph // показва ти като граф 
	
	// ако искаш сам да си зададеш в какъв формат искаш да ти излезе историята за git 
	git log --format="%h - %an, %ar : %s"
	
	// ако искаш да видиш комитите последните две седмици
	git log --since=2.weeks
	
	// ако искаш да търсиш за ключова дума между комитите 
	git log -S <functionName>
